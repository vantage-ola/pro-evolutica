cmake_minimum_required(VERSION 3.22)
project(pro_evolutica LANGUAGES CXX)

if(ANDROID)
    find_package(SDL3 REQUIRED CONFIG)
    add_library(main SHARED app/main.cpp rendering/cube.cpp rendering/cube_data.cpp rendering/camera.cpp assets/png.cpp assets/mesh.cpp)
    target_compile_features(main PRIVATE cxx_std_17)
    target_compile_options(main PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(main PRIVATE SDL3::SDL3 GLESv3 z)
else()
    enable_testing()
    add_executable(cube_data_tests tests/cube_data_test.cpp rendering/cube_data.cpp)
    target_compile_features(cube_data_tests PRIVATE cxx_std_17)
    target_compile_options(cube_data_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME cube_data COMMAND cube_data_tests)
    add_executable(camera_tests tests/camera_test.cpp rendering/camera.cpp rendering/cube_data.cpp)
    target_compile_features(camera_tests PRIVATE cxx_std_17)
    target_compile_options(camera_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME camera COMMAND camera_tests)
    find_package(ZLIB REQUIRED)
    add_executable(png_tests tests/png_test.cpp assets/png.cpp)
    target_compile_features(png_tests PRIVATE cxx_std_17)
    target_compile_options(png_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(png_tests PRIVATE ZLIB::ZLIB)
    add_test(NAME png COMMAND png_tests)
    set_tests_properties(png PROPERTIES TIMEOUT 10)
    add_executable(mesh_tests tests/mesh_test.cpp assets/mesh.cpp rendering/cube_data.cpp)
    target_compile_features(mesh_tests PRIVATE cxx_std_17)
    target_compile_options(mesh_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME mesh COMMAND mesh_tests)
    set_tests_properties(mesh PROPERTIES TIMEOUT 10)
endif()
